import { useState } from "react";
import { HashRouter, Routes, Route, Navigate } from "react-router-dom";
import { AppProvider } from "./hooks/useAppState";
import Sidebar from "./components/Sidebar";
import Topbar from "./components/Topbar";
import EvidencePanel from "./components/EvidencePanel";
import { DisclaimerBar } from "./components/ui";
import Dashboard from "./pages/Dashboard";
import Reports from "./pages/Reports";
import Signals from "./pages/Signals";
import ChangeIntelligence from "./pages/ChangeIntelligence";
import RiskIntelligence from "./pages/RiskIntelligence";
import Auditor from "./pages/Auditor";
import AskReport from "./pages/AskReport";
import Evidence from "./pages/Evidence";
import AuditLog from "./pages/AuditLog";
import Settings from "./pages/Settings";
import Upload from "./pages/Upload";

export default function App() {
  const [sidebarOpen, setSidebarOpen] = useState(false);

  return (
    <AppProvider>
      <HashRouter>
        <div className="min-h-screen">
          <div className="flex">
            <Sidebar open={sidebarOpen} onClose={() => setSidebarOpen(false)} />
            <div className="min-w-0 flex-1">
              <Topbar onMenu={() => setSidebarOpen(true)} />
              <main className="mx-auto max-w-7xl px-4 py-6 pb-16 lg:px-8">
                <Routes>
                  <Route path="/" element={<Navigate to="/dashboard" replace />} />
                  <Route path="/dashboard" element={<Dashboard />} />
                  <Route path="/reports" element={<Reports />} />
                  <Route path="/signals" element={<Signals />} />
                  <Route path="/change" element={<ChangeIntelligence />} />
                  <Route path="/risk" element={<RiskIntelligence />} />
                  <Route path="/auditor" element={<Auditor />} />
                  <Route path="/ask" element={<AskReport />} />
                  <Route path="/evidence" element={<Evidence />} />
                  <Route path="/audit-log" element={<AuditLog />} />
                  <Route path="/settings" element={<Settings />} />
                  <Route path="/upload" element={<Upload />} />
                  <Route path="*" element={<Navigate to="/dashboard" replace />} />
                </Routes>
              </main>
            </div>
          </div>
          <EvidencePanel />
          <DisclaimerBar />
        </div>
      </HashRouter>
    </AppProvider>
  );
}

export const pctChange = (prev, curr) => Math.round(((curr - prev) / prev) * 1000) / 10;

export const isMaterial = (deltaPct, threshold = 15) => Math.abs(deltaPct) > threshold;

export function classify(deltaPct, goodWhenUp = true) {
  const up = deltaPct >= 0;
  const good = up ? goodWhenUp : !goodWhenUp;
  return { up, good, tone: good ? "good" : "bad", color: good ? "text-emerald-600" : "text-rose-600" };
}

export function AppProvider({ children }) {
  const [role, setRole] = useState("analyst");
  const [compressionLevel, setCompressionLevel] = useState("c75");
  const [evidenceId, setEvidenceId] = useState(null);
  const [watchlist, setWatchlist] = useState(["AL-02"]);

  const openEvidence = useCallback((id) => setEvidenceId(id), []);
  const closeEvidence = useCallback(() => setEvidenceId(null), []);
  const toggleWatch = useCallback(
    (id) => setWatchlist((w) => (w.includes(id) ? w.filter((x) => x !== id) : [...w, id])), []);

  const value = useMemo(
    () => ({ role, setRole, compressionLevel, setCompressionLevel, evidenceId, openEvidence, closeEvidence, watchlist, toggleWatch }),
    [role, compressionLevel, evidenceId, watchlist, openEvidence, closeEvidence, toggleWatch]);

  return <AppCtx.Provider value={value}>{children}</AppCtx.Provider>;
}

  return ctx;

import { useEffect, useRef, useState } from "react";

export function usePipeline(steps, { stepMs = 850 } = {}) {
  const [active, setActive] = useState(-1);
  const timer = useRef(null);
  const running = active >= 0 && active < steps.length;
  const done = active >= steps.length;

  useEffect(() => {
    if (running) {
      timer.current = setTimeout(() => setActive((a) => a + 1), stepMs);
      return () => clearTimeout(timer.current);
    }
  }, [active, running, stepMs]);

  return {
    active, running, done,
    start: () => setActive(0),
    reset: () => setActive(-1),
    status: (i) => (done || active > i ? "done" : active === i ? "active" : "pending"),
  };
}

export default function Sparkline({ data = [], stroke = "#3A5CA0", width = 72, height = 26 }) {
  if (data.length < 2) return null;
  const min = Math.min(...data), max = Math.max(...data), range = max - min || 1;
  const pts = data
    .map((v, i) => `${(i / (data.length - 1)) * (width - 6) + 3},${height - 4 - ((v - min) / range) * (height - 8)}`)
    .join(" ");
  const last = pts.split(" ").pop().split(",");
  return (
    <svg width={width} height={height} className="overflow-visible">
      <polyline points={pts} fill="none" stroke={stroke} strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" />
      <circle cx={last[0]} cy={last[1]} r="2.5" fill={stroke} />
    </svg>
  );
}

  if (!active || !payload?.length) return null;
  const q1 = payload.find((p) => p.dataKey === "Q1")?.value;

    <div className="px-3 py-2" style={TIP}>

}
